const mongoose = require('mongoose')
const { Schema } = mongoose

const UserSchema = new Schema(
  {
    email: { type: String, required: true, unique: true },
    name: { type: String, required: true, default: "anon" },
    age: { type: Number, default: 18 },
    score: { type: Number, default: 1.5 },
    active: { type: Boolean, default: true },
    created_at: { type: Date, default: Date.now },
    address: {
      street: { type: String, required: true },
      geo: {
        lat: { type: Number, required: true },
        lng: { type: Number },
        deep: {
          a: {
            b: {
              c: { type: String, required: true },
            },
          },
        },
      },
      lines: [{
        sku: { type: String },
        qty: { type: Number, required: true },
      }],
    },
    tags: [{ type: String }],
    parent_id: { type: Schema.Types.ObjectId, ref: 'User' },
    manager_id: { type: Schema.Types.ObjectId, ref: 'User' },
    bio: { type: String },
    avatar: { type: Buffer },
    meta: { type: Schema.Types.Mixed },
    raw: { type: Schema.Types.Mixed },
    amount: { type: Number },
    big: { type: Number },
    tags_ids: [{ type: Schema.Types.ObjectId, ref: 'Tags' }],
  },
  { timestamps: false },
)
UserSchema.index({ name: 1, age: 1 }, { unique: true })
UserSchema.index({ email: 1 })
const User = mongoose.model('User', UserSchema, 'user')

const OrderSchema = new Schema(
  {
    user_id: { type: Schema.Types.ObjectId, ref: 'User', required: true },
    items: [{
      sku: { type: String, required: true },
      price: { type: Number },
    }],
    order_item: {
      sku: { type: String, required: true },
      price: { type: Number },
    },
    status: { type: String, default: "new" },
  },
  { timestamps: false },
)
const Order = mongoose.model('Order', OrderSchema, 'order')

const OrderItemSchema = new Schema(
  {
    order_id: { type: Schema.Types.ObjectId, ref: 'Order' },
  },
  { timestamps: false },
)
const OrderItem = mongoose.model('OrderItem', OrderItemSchema, 'order_item')

const TagsSchema = new Schema(
  {
    label: { type: String },
  },
  { timestamps: false },
)
const Tags = mongoose.model('Tags', TagsSchema, 'tags')

const CategorySchema = new Schema(
  {
    parent_id: { type: Schema.Types.ObjectId, ref: 'Category' },
    children: [Schema.Types.Mixed],
  },
  { timestamps: false },
)
const Category = mongoose.model('Category', CategorySchema, 'category')

module.exports = { User, Order, OrderItem, Tags, Category }
